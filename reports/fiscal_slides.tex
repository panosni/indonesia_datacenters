% ─── SLIDE 3: ASSUMPTIONS ──────────────────────────────────────────────────────
\begin{frame}{\textbf{Key Assumptions}}
\vspace{0.1cm}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\color{navyblue}A.\ Investment values}\\[2pt]
    {\footnotesize Announced USD figures converted at USD/IDR 15,500. MW-based imputation used where monetary data absent (USD 1.5M/MW operational, USD 2M/MW planned).}\\[8pt]

    \textbf{\color{navyblue}B.\ Profitability}\\[2pt]
    {\footnotesize Annual EBIT = 12\% of total investment. Applied uniformly as a sector benchmark for fiscal cost computation.}\\[8pt]

    \textbf{\color{navyblue}C.\ Tax Holiday (PMK 69/2024)}\\[2pt]
    {\footnotesize CIT = 22\%. Tiers: $\geq$IDR 100T $\to$ 20 yrs; IDR 30T–100T $\to$ 15 yrs; IDR 500B–30T $\to$ 10 yrs; IDR 100B–500B $\to$ 5 yrs; $<$IDR 100B $\to$ 0 yrs. 50\% reduction for 2-yr extensions where applicable.}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{\color{navyblue}D.\ BEPS Pillar Two}\\[2pt]
    {\footnotesize In scope: global group revenue $>$€750M AND operating/EBITDA margin $>$10\%. GloBE minimum rate = 15\%. P2 in-scope: Indonesia forfeits 7\% (22\%$-$15\%). P2 out-of-scope: Indonesia forfeits full 22\% (or 11\% for 50\% reduction facilities).}\\[8pt]

    \textbf{\color{navyblue}E.\ Super Deduction (PP 78/2019)}\\[2pt]
    {\footnotesize 30\% net income reduction over 6 years at 5\%/yr. Applies after tax holiday ends. Data centres qualify under KBLI 63111. Fiscal cost = deduction $\times$ CIT-forfeited rate $\times$ 6 yrs. Annual deduction = 5\% of investment.}\\[8pt]

    \textbf{\color{navyblue}F.\ SEZ VAT Exemption}\\[2pt]
    {\footnotesize 4 facilities in Nongsa Digital Park KEK (Batam). VAT cost = investment $\times$ 35\% $\times$ 11\% (PMK-131/2024: 12\% $\times$ 11/12).}
  \end{column}
\end{columns}
\end{frame}
